function Avatar({ member, size = 26 }) {
  return (
    <div className="avatar" style={{ width: size, height: size, background: member.color + "33", color: member.color, fontSize: size * 0.35 }}>
      {member.avatar}
    </div>
  );
}

function PriorityBadge({ priority }) {
  const p = PRIORITIES[priority];
  return <span className="priority-badge" style={{ color: p.color, background: p.bg }}>{p.label}</span>;
}

function TaskCard({ task, onClick, onDragStart, onDragEnd }) {
  const member = MEMBERS.find(m => m.id === task.assignee);
  const isOverdue = new Date(task.due) < new Date() && task.due;
  return (
    <div
      className="task-card"
      onClick={() => onClick(task)}
      draggable
      onDragStart={e => { e.dataTransfer.setData("taskId", task.id); onDragStart?.(); }}
      onDragEnd={onDragEnd}
    >
      <div className="task-header">
        <div className="task-title">{task.title}</div>
        <PriorityBadge priority={task.priority} />
      </div>
      <div className="task-desc">{task.desc}</div>
      {task.tags.length > 0 && (
        <div className="task-tags">
          {task.tags.map(t => <span key={t} className="tag">{t}</span>)}
        </div>
      )}
      <div className="task-footer">
        {member && <Avatar member={member} size={22} />}
        {task.comments.length > 0 && (
          <div className="comment-count">💬 {task.comments.length}</div>
        )}
        {task.due && (
          <div className={`task-due ${isOverdue ? "overdue" : ""}`}>
            {isOverdue ? "⚠ " : ""}{task.due}
          </div>
        )}
      </div>
    </div>
  );
}

function Column({ col, tasks, onTaskClick, onDrop }) {
  const [dragOver, setDragOver] = useState(false);
  const [dragging, setDragging] = useState(false);
  return (
    <div className="column">
      <div className="col-header">
        <span className="col-icon" style={{ color: col.color }}>{col.icon}</span>
        <span className="col-title">{col.label}</span>
        <span className="col-count">{tasks.length}</span>
      </div>
      <div
        className="col-body"
        onDragOver={e => { e.preventDefault(); setDragOver(true); }}
        onDragLeave={() => setDragOver(false)}
        onDrop={e => { e.preventDefault(); setDragOver(false); const id = parseInt(e.dataTransfer.getData("taskId")); onDrop(id, col.id); }}
      >
        {tasks.map(t => (
          <TaskCard key={t.id} task={t} onClick={onTaskClick}
            onDragStart={() => setDragging(true)}
            onDragEnd={() => setDragging(false)}
          />
        ))}
        <div className={`drop-zone ${dragOver ? "active" : ""}`}>
          {dragOver ? "Drop here" : "+ Drop task"}
        </div>
      </div>
    </div>
  );
}

function TaskModal({ task, onClose, onAddComment, onMove }) {
  const member = MEMBERS.find(m => m.id === task.assignee);
  const [comment, setComment] = useState("");
  const handleComment = () => {
    if (!comment.trim()) return;
    onAddComment(task.id, comment);
    setComment("");
  };
  return (
    <div className="modal-overlay" onClick={e => e.target === e.currentTarget && onClose()}>
      <div className="modal">
        <div className="modal-header">
          <div>
            <div style={{ display: "flex", gap: 8, alignItems: "center", marginBottom: 6 }}>
              <PriorityBadge priority={task.priority} />
              {task.tags.map(t => <span key={t} className="tag">{t}</span>)}
            </div>
            <div className="modal-title">{task.title}</div>
          </div>
          <button className="modal-close" onClick={onClose}>✕</button>
        </div>
        <div className="modal-body">
          <div className="modal-section">
            <div className="modal-label">Description</div>
            <div style={{ fontSize: 13.5, color: "#94a3b8", lineHeight: 1.6 }}>{task.desc}</div>
          </div>
          <div className="modal-row" style={{ marginBottom: 16 }}>
            <div>
              <div className="modal-label">Assignee</div>
              {member && <div style={{ display: "flex", alignItems: "center", gap: 8 }}>
                <Avatar member={member} size={28} />
                <span style={{ fontSize: 13 }}>{member.name}</span>
              </div>}
            </div>
            <div>
              <div className="modal-label">Due Date</div>
              <div style={{ fontSize: 13, color: "#94a3b8" }}>{task.due || "No due date"}</div>
            </div>
            <div>
              <div className="modal-label">Move To</div>
              <select className="modal-input select-input" style={{ fontSize: 12, padding: "6px 10px" }}
                onChange={e => { onMove(task.id, e.target.value); onClose(); }}>
                <option value="">— Move —</option>
                {COLUMNS.map(c => <option key={c.id} value={c.id}>{c.label}</option>)}
              </select>
            </div>
          </div>
          <div className="modal-section">
            <div className="modal-label">Comments ({task.comments.length})</div>
            {task.comments.length > 0 && (
              <div className="comments-list">
                {task.comments.map((c, i) => (
                  <div key={i} className="comment">
                    <div className="comment-header">
                      <span className="comment-author">{c.user}</span>
                      <span className="comment-time">{c.time}</span>
                    </div>
                    <div className="comment-text">{c.text}</div>
                  </div>
                ))}
              </div>
            )}
            <div className="comment-input-row">
              <input className="comment-input" placeholder="Add a comment…" value={comment}
                onChange={e => setComment(e.target.value)}
                onKeyDown={e => e.key === "Enter" && handleComment()} />
              <button className="btn btn-primary btn-sm" onClick={handleComment}>Send</button>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
}

function AddTaskModal({ onClose, onAdd }) {
  const [form, setForm] = useState({ title: "", desc: "", priority: "medium", assignee: 1, due: "", tags: "", col: "todo" });
  const set = (k, v) => setForm(f => ({ ...f, [k]: v }));
  const handleAdd = () => {
    if (!form.title.trim()) return;
    onAdd({
      id: Date.now(),
      title: form.title,
      desc: form.desc,
      priority: form.priority,
      assignee: parseInt(form.assignee),
      due: form.due,
      comments: [],
      tags: form.tags.split(",").map(t => t.trim()).filter(Boolean),
    }, form.col);
    onClose();
  };
  return (
    <div className="modal-overlay" onClick={e => e.target === e.currentTarget && onClose()}>
      <div className="modal">
        <div className="modal-header">
          <div className="modal-title">New Task</div>
          <button className="modal-close" onClick={onClose}>✕</button>
        </div>
        <div className="modal-body">
          <div className="new-task-form">
            <div>
              <div className="modal-label">Title *</div>
              <input className="modal-input" placeholder="Task title…" value={form.title} onChange={e => set("title", e.target.value)} />
            </div>
            <div>
              <div className="modal-label">Description</div>
              <textarea className="modal-input" placeholder="What needs to be done…" value={form.desc} onChange={e => set("desc", e.target.value)} />
            </div>
            <div className="modal-row">
              <div>
                <div className="modal-label">Priority</div>
                <select className="modal-input select-input" value={form.priority} onChange={e => set("priority", e.target.value)}>
                  <option value="high">High</option>
                  <option value="medium">Medium</option>
                  <option value="low">Low</option>
                </select>
              </div>
              <div>
                <div className="modal-label">Column</div>
                <select className="modal-input select-input" value={form.col} onChange={e => set("col", e.target.value)}>
                  {COLUMNS.map(c => <option key={c.id} value={c.id}>{c.label}</option>)}
                </select>
              </div>
            </div>
            <div className="modal-row">
              <div>
                <div className="modal-label">Assignee</div>
                <select className="modal-input select-input" value={form.assignee} onChange={e => set("assignee", e.target.value)}>
                  {MEMBERS.map(m => <option key={m.id} value={m.id}>{m.name}</option>)}
                </select>
              </div>
              <div>
                <div className="modal-label">Due Date</div>
                <input className="modal-input" type="date" value={form.due} onChange={e => set("due", e.target.value)} />
              </div>
            </div>
            <div>
              <div className="modal-label">Tags (comma-separated)</div>
              <input className="modal-input" placeholder="Frontend, Backend…" value={form.tags} onChange={e => set("tags", e.target.value)} />
            </div>
            <div style={{ display: "flex", gap: 10, justifyContent: "flex-end" }}>
              <button className="btn btn-ghost" onClick={onClose}>Cancel</button>
              <button className="btn btn-primary" onClick={handleAdd}>Create Task</button>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
}

const NOTIFICATIONS = [
  { text: "Sara M. commented on Kanban board UI", time: "5m ago" },
  { text: "3 tasks are overdue — check In Progress", time: "1h ago" },
  { text: "Priya R. moved OAuth integration to Review", time: "2h ago" },
  { text: "New team member John D. joined the project", time: "1d ago" },
];

export default function App() {
  const [tasks, setTasks] = useState(INITIAL_TASKS);
  const [selectedTask, setSelectedTask] = useState(null);
  const [showAdd, setShowAdd] = useState(false);
  const [showNotif, setShowNotif] = useState(false);
  const [search, setSearch] = useState("");
  const [activeNav, setActiveNav] = useState("board");

  const moveTask = (taskId, toCol) => {
    setTasks(prev => {
      const next = {};
      let found = null;
      for (const col in prev) {
        next[col] = prev[col].filter(t => { if (t.id === taskId) { found = t; return false; } return true; });
      }
      if (found) next[toCol] = [...next[toCol], found];
      return next;
    });
  };

  const addComment = (taskId, text) => {
    setTasks(prev => {
      const next = {};
      for (const col in prev) {
        next[col] = prev[col].map(t => t.id === taskId
          ? { ...t, comments: [...t.comments, { user: "You", text, time: "just now" }] }
          : t);
      }
      return next;
    });
    if (selectedTask?.id === taskId) {
      setSelectedTask(t => ({ ...t, comments: [...t.comments, { user: "You", text, time: "just now" }] }));
    }
  };

  const addTask = (task, col) => {
    setTasks(prev => ({ ...prev, [col]: [...prev[col], task] }));
  };

  const filteredTasks = (col) => {
    if (!search) return tasks[col];
    return tasks[col].filter(t => t.title.toLowerCase().includes(search.toLowerCase()) || t.desc.toLowerCase().includes(search.toLowerCase()));
  };

  const total = Object.values(tasks).flat().length;
  const done = tasks.done.length;
  const inprog = tasks.inprogress.length;
  const overdue = Object.values(tasks).flat().filter(t => t.due && new Date(t.due) < new Date()).length;

  return (
    <div className="app">
      {/* SIDEBAR */}
      <div className="sidebar">
        <div className="logo">
          <div className="logo-text">TaskFlow</div>
          <div className="logo-sub">Team Workspace</div>
        </div>
        <div className="nav">
          <div className="nav-section">Workspace</div>
          {[
            { id: "board", icon: "⊞", label: "Board" },
            { id: "tasks", icon: "✓", label: "My Tasks", badge: 3 },
            { id: "calendar", icon: "◻", label: "Calendar" },
            { id: "reports", icon: "↗", label: "Reports" },
          ].map(n => (
            <div key={n.id} className={`nav-item ${activeNav === n.id ? "active" : ""}`} onClick={() => setActiveNav(n.id)}>
              <span className="icon">{n.icon}</span>
              {n.label}
              {n.badge && <span className="nav-badge">{n.badge}</span>}
            </div>
          ))}
          <div className="nav-section" style={{ marginTop: 8 }}>Tools</div>
          {[
            { id: "notif", icon: "🔔", label: "Notifications", badge: 4 },
            { id: "settings", icon: "⚙", label: "Settings" },
          ].map(n => (
            <div key={n.id} className="nav-item" onClick={() => n.id === "notif" && setShowNotif(v => !v)}>
              <span className="icon">{n.icon}</span>
              {n.label}
              {n.badge && <span className="nav-badge">{n.badge}</span>}
            </div>
          ))}
        </div>
        <div className="team-section">
          <div className="team-label">Team Online</div>
          {MEMBERS.map(m => (
            <div key={m.id} className="team-member">
              <Avatar member={m} size={26} />
              <span className="member-name">{m.name}</span>
              <div className="online-dot" />
            </div>
          ))}
        </div>
      </div>

      {/* MAIN */}
      <div className="main">
        <div className="header">
          <div className="header-title">Collaborative Task <span>Management</span></div>
          <div className="search-box">
            <span style={{ color: "#64748b", fontSize: 13 }}>⌕</span>
            <input placeholder="Search tasks…" value={search} onChange={e => setSearch(e.target.value)} />
          </div>
          <button className="btn btn-ghost" onClick={() => setShowNotif(v => !v)}>🔔 4</button>
          <button className="btn btn-primary" onClick={() => setShowAdd(true)}>+ New Task</button>
        </div>

        {/* STATS */}
        <div className="stats-bar">
          <div className="stat-card">
            <div className="stat-num" style={{ color: COLORS.accentLight }}>{total}</div>
            <div className="stat-label">Total Tasks</div>
            <div className="stat-change up">↑ +2 this week</div>
          </div>
          <div className="stat-card">
            <div className="stat-num" style={{ color: COLORS.blue }}>{inprog}</div>
            <div className="stat-label">In Progress</div>
            <div className="stat-change up">Active sprints</div>
          </div>
          <div className="stat-card">
            <div className="stat-num" style={{ color: COLORS.green }}>{done}</div>
            <div className="stat-label">Completed</div>
            <div className="stat-change up">↑ {Math.round((done / total) * 100)}% done</div>
          </div>
          <div className="stat-card">
            <div className="stat-num" style={{ color: overdue > 0 ? COLORS.red : COLORS.green }}>{overdue}</div>
            <div className="stat-label">Overdue</div>
            <div className={`stat-change ${overdue > 0 ? "warn" : "up"}`}>{overdue > 0 ? "⚠ Needs attention" : "All on track"}</div>
          </div>
          <div className="stat-card">
            <div className="stat-num" style={{ color: COLORS.pink }}>{MEMBERS.length}</div>
            <div className="stat-label">Team Members</div>
            <div className="stat-change up">All online</div>
          </div>
        </div>

        {/* BOARD */}
        <div className="board-wrap">
          <div className="board">
            {COLUMNS.map(col => (
              <Column
                key={col.id}
                col={col}
                tasks={filteredTasks(col.id)}
                onTaskClick={t => setSelectedTask(t)}
                onDrop={moveTask}
              />
            ))}
          </div>
        </div>
      </div>

      {/* NOTIFICATIONS */}
      {showNotif && (
        <div className="notif-panel">
          <div className="notif-title">🔔 Notifications</div>
          {NOTIFICATIONS.map((n, i) => (
            <div key={i} className="notif-item">
              <div className="notif-dot" />
              <div>
                <div className="notif-text">{n.text}</div>
                <div className="notif-time">{n.time}</div>
              </div>
            </div>
          ))}
        </div>
      )}

      {/* TASK DETAIL MODAL */}
      {selectedTask && (
        <TaskModal
          task={selectedTask}
          onClose={() => setSelectedTask(null)}
          onAddComment={addComment}
          onMove={moveTask}
        />
      )}

      {/* ADD TASK MODAL */}
      {showAdd && (
        <AddTaskModal onClose={() => setShowAdd(false)} onAdd={addTask} />
      )}
    </div>
  );
}
